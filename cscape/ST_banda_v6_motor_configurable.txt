(* MASTER BANDA - HORNER / CSCAPE ST - v6 MOTOR CONFIGURABLE   *)
(*                                                            *)
(* v6 = v5 INTACTO + un MOTOR CONFIGURABLE (secciones §E).     *)
(*   EngineMode = EC[0] = %R200                                *)
(*     0 -> se ejecuta la logica v5 de siempre (§5, §11..§17). *)
(*     1 -> el motor configurable controla VFD, torreta y      *)
(*          plumas; v5 queda en espera.                        *)
(*   Configuracion : EC[0..259] = %R200..%R459 (INT, escribe   *)
(*                   el backend: banda_motor.py)               *)
(*   Estado        : ES[0..39]  = %R460..%R499 (INT, solo      *)
(*                   lectura; ES[0] = 6 identifica el v6)      *)
(*   Base de tiempo: T_100MS (%S4), resolucion 0.1 s.           *)
(* Comparte con v5: §0 arranque seguro, §3 validacion, §4      *)
(* paros (I3 SIEMPRE manda), §6/§7 NewCfgFlag/ResetCmd, §8/§9  *)
(* preparacion y frecuencia del VFD, §18..§20 feedback.        *)
(* ============================================================ *)
(*                                                            *)
(*                                                            *)
(* OBJETIVO v5:                                               *)
(* - S1/S2 son eventos generales independientes de la banda.  *)
(* - Cada deteccion puede ejecutar acciones inmediatas.        *)
(* - El contador trabaja EN PARALELO al evento inmediato.      *)
(* - Al alcanzar CountPreset puede combinar: pausa de banda,   *)
(*   stop de proceso, luces, direccion y ambas plumas.         *)
(* - Las acciones temporales del contador pueden ser           *)
(*   enclavadas o temporizadas.                                *)
(* - Existe una lampara temporizada independiente.             *)
(*                                                            *)
(* DirCmd:                                                     *)
(*   0 = sin movimiento                                        *)
(*   1 = Direccion 1 / derecha -> VFD control 18              *)
(*   2 = Direccion 2 / izquierda -> VFD control 34            *)
(*                                                            *)
(* StopMode (%R9):                                             *)
(*   0 = I3 solamente                                         *)
(*   1 = I2 + I3                                              *)
(*   2 = Stop software/enclavamiento + I3                     *)
(*   3 = I2 + Stop software/enclavamiento + I3                *)
(* I3 SIEMPRE tiene prioridad y no se puede deshabilitar.     *)
(*                                                            *)
(* Sensor Action (accion inmediata de CADA deteccion):         *)
(*   0 = evento sin parar banda                               *)
(*   1 = pausa banda mientras sensor activo                    *)
(*   2 = pausa/evento por tiempo                               *)
(*   3 = igual a 1; torreta se define por TorretaMask         *)
(*   4 = igual a 2; torreta se define por TorretaMask         *)
(* Sensor BandMode: 0=Action puede pausar banda;               *)
(*                  1=evento auxiliar sin pausar banda.        *)
(*                                                            *)
(* IMPORTANTE v5: CountPreset YA NO bloquea la accion          *)
(* inmediata. Ejemplo: S1 puede subir una pluma en cada pieza  *)
(* y, simultaneamente, al conteo 3 encender una luz.           *)
(*                                                            *)
(* CountActionMask R70/R75:                                   *)
(*   1  = pausar banda por contador                            *)
(*   2  = detener proceso completo (enclavado hasta R5/R6)     *)
(*   4  = luces segun CountLampMask                            *)
(*   8  = cambiar direccion de forma segura                    *)
(*   16 = comando Pluma 1                                      *)
(*   32 = comando Pluma 2                                      *)
(*   Se SUMAN para combinar acciones.                          *)
(*                                                            *)
(* CountHoldMode R88/R92:                                     *)
(*   0 = acciones temporales quedan enclavadas hasta R5/R6     *)
(*   1 = acciones temporales duran CountHoldPreset segundos    *)
(* Aplica a pausa de banda + luces + plumas del contador.      *)
(* El stop total de proceso permanece enclavado por seguridad  *)
(* de proceso; el cambio de direccion es persistente.          *)
(*                                                            *)
(* Lampara temporizada independiente:                          *)
(*   R96=mask, R97=segundos, R98=trigger 1,2,3..., R99=active. *)
(*                                                            *)
(* TorretaMask: bit0 verde, bit1 amarilla, bit2 roja (0..7).   *)
(* Las fuentes de luz se SUMAN: RUN/IDLE + S1 + S2 + I1 +      *)
(* contador + lampara temporizada. I3 apaga todas.             *)
(*                                                            *)
(* PlumaCmd: 0=Stop, 1=Subir, 2=Bajar; sensor/count acepta 3=stop. *)
(* Mapeo fisico confirmado:                                   *)
(*   Pluma 1: Q8 SUBE / Q9 BAJA                               *)
(*   Pluma 2: Q6 SUBE / Q7 BAJA                               *)
(* ============================================================ *)

(* ============================================================ *)
(* §3b - VALIDACION DEL MOTOR CONFIGURABLE (v6)                *)
(* Espejo de banda_motor.error_registros(). Si un campo de EC  *)
(* esta fuera de rango, CfgValid = FALSE y ES[2] = offset + 1  *)
(* del primer campo invalido (el backend lo traduce a texto).  *)
(* Codigo de senal valido: 0..63 o 100..163 (NOT).             *)
(* ============================================================ *)

EngineMode := EC[0] ;
EngValid   := TRUE ;
EngError   := 0 ;

IF (EngineMode < 0) OR (EngineMode > 1) THEN

    EngValid := FALSE ;
    EngError := 1 ;

ELSIF EngineMode = 1 THEN

    FOR ek := 1 TO 259 DO

        esrc := FALSE ;
        echk := FALSE ;
        elo  := 0 ;
        ehi  := 32767 ;

        IF (ek = 1) OR (ek = 2) OR ((ek >= 4) AND (ek <= 7)) THEN
            esrc := TRUE ; echk := TRUE ;
        ELSIF (ek = 3) OR (ek = 11) OR (ek = 12) OR (ek = 15) THEN
            echk := TRUE ; ehi := 1 ;
        ELSIF (ek = 8) OR (ek = 9) OR (ek = 10) OR (ek = 13) THEN
            echk := TRUE ;
        ELSIF ek = 14 THEN
            echk := TRUE ; ehi := 7 ;
        ELSIF (ek >= 20) AND (ek <= 26) THEN
            esrc := TRUE ; echk := TRUE ;
        ELSIF (ek >= 30) AND (ek <= 77) THEN
            er := (ek - 30) MOD 6 ;
            IF (er >= 1) AND (er <= 4) THEN
                esrc := TRUE ; echk := TRUE ;
            ELSE
                echk := TRUE ; ehi := 2 ;
            END_IF ;
        ELSIF (ek >= 78) AND (ek <= 109) THEN
            er := (ek - 78) MOD 4 ;
            IF (er = 1) OR (er = 2) THEN
                esrc := TRUE ; echk := TRUE ;
            ELSIF er = 0 THEN
                echk := TRUE ; ehi := 4 ;
            ELSE
                echk := TRUE ;
            END_IF ;
        ELSIF (ek >= 110) AND (ek <= 141) THEN
            er := (ek - 110) MOD 4 ;
            IF er <= 1 THEN
                esrc := TRUE ; echk := TRUE ;
            ELSIF er = 2 THEN
                echk := TRUE ;
            ELSE
                echk := TRUE ; ehi := 1 ;
            END_IF ;
        ELSIF (ek >= 142) AND (ek <= 165) THEN
            er := (ek - 142) MOD 3 ;
            IF er <= 1 THEN
                esrc := TRUE ; echk := TRUE ;
            ELSE
                echk := TRUE ; ehi := 1 ;
            END_IF ;
        ELSIF (ek >= 166) AND (ek <= 168) THEN
            esrc := TRUE ; echk := TRUE ;
        ELSIF ek = 169 THEN
            echk := TRUE ; ehi := 1 ;
        ELSIF ek = 170 THEN
            echk := TRUE ; ehi := 16 ;
        ELSIF (ek >= 172) AND (ek <= 251) THEN
            er := (ek - 172) MOD 5 ;
            IF er = 2 THEN
                esrc := TRUE ; echk := TRUE ;
            ELSIF er = 0 THEN
                echk := TRUE ; ehi := 4095 ;
            ELSIF er = 1 THEN
                echk := TRUE ; ehi := 4 ;
            ELSIF er = 3 THEN
                echk := TRUE ;
            ELSE
                echk := TRUE ; ehi := 16 ;
            END_IF ;
        END_IF ;

        IF echk AND EngValid THEN
            ev := EC[ek] ;
            IF esrc THEN
                IF NOT (((ev >= 0) AND (ev <= 63)) OR ((ev >= 100) AND (ev <= 163))) THEN
                    EngValid := FALSE ;
                    EngError := ek + 1 ;
                END_IF ;
            ELSIF (ev < elo) OR (ev > ehi) THEN
                EngValid := FALSE ;
                EngError := ek + 1 ;
            END_IF ;
        END_IF ;

    END_FOR ;

END_IF ;

IF (EngineMode = 1) AND NOT EngValid THEN

END_IF ;


(* I3 es el paro duro fisico.

IF SystemStopActive THEN

    S1_CountAccum   := 0 ;

(* §8 - MAQUINA DE ESTADOS: PREPARACION / RESET VFD           *)
(* ============================================================ *)

IF NOT CfgValid THEN

    CfgReady     := FALSE ;
    CfgReady_Reg := 0 ;

    InitSeqState := 0 ;

    VFD_Control  := 1 ;
    VFD_ResetReg := 0 ;
    VFD_FreqCalc := 0 ;


ELSIF NOT MotionCfgPresent THEN

    (* Configuracion valida sin movimiento: sensores/torreta/plumas
       pueden trabajar sin inicializar ni ejecutar el VFD. *)

    CfgReady      := TRUE ;
    CfgReady_Reg  := 1 ;
    InitSeqState  := 0 ;

    VFD_Control   := 1 ;
    VFD_ResetReg  := 0 ;
    VFD_FreqCalc  := 0 ;


ELSIF NOT GenStop THEN


    (* Estado 0 = VFD listo / sin inicializacion activa *)

    IF InitSeqState = 0 THEN

        VFD_ResetReg := 0 ;


    (* Estado 1 = garantizar STOP *)

    ELSIF InitSeqState = 1 THEN

        VFD_Control    := 1 ;
        VFD_ResetReg   := 0 ;

        InitTimerAccum := 0 ;
        InitSeqState   := 2 ;


    (* Estado 2 = activar reset VFD *)

    ELSIF InitSeqState = 2 THEN

        VFD_Control    := 1 ;
        VFD_ResetReg   := 2 ;

        InitTimerAccum := 0 ;
        InitSeqState   := 3 ;


    (* Estado 3 = mantener reset durante 2 segundos *)

    ELSIF InitSeqState = 3 THEN

        VFD_Control  := 1 ;
        VFD_ResetReg := 2 ;

        IF Pulse1s THEN
            InitTimerAccum := InitTimerAccum + 1 ;
        END_IF ;

        IF InitTimerAccum >= 2 THEN
            InitSeqState := 4 ;
        END_IF ;


    (* Estado 4 = liberar reset y cargar frecuencia *)

    ELSIF InitSeqState = 4 THEN

        VFD_ResetReg := 0 ;
        VFD_Control  := 1 ;

        VFD_FreqCalc :=
            FreqRequest * INT#100 ;

        CfgReady     := TRUE ;
        CfgReady_Reg := 1 ;

        InitSeqState := 0 ;


    ELSE

        InitSeqState := 0 ;

    END_IF ;


ELSE

    (* STOP presionado *)

    VFD_Control  := 1 ;
    VFD_ResetReg := 0 ;

END_IF ;

(* El backend trabaja en Hz; el VFD recibe Hz x100.           *)
(* Ejemplo: 30 Hz -> 3000.                                    *)
(* ============================================================ *)

IF MotionCfgPresent AND
   CfgReady AND
   CfgValid AND
   (InitSeqState = 0) AND
   NOT GenStop THEN

    VFD_FreqCalc :=
        FreqRequest * INT#100 ;

END_IF ;

(* ============================================================ *)
(* v6: §11..§17 son la logica v5. Solo corren con EngineMode=0. *)
(* Con EngineMode=1 corre el MOTOR CONFIGURABLE (§E1..§E10).   *)
(* ============================================================ *)

IF EngineMode = 0 THEN


(* ============================================================ *)
(* §11 - SENSOR S1 COMO EVENTO GENERAL                         *)
(*                                                            *)
(* v5: la accion inmediata y el contador son INDEPENDIENTES.   *)
(* CADA flanco de S1 puede activar torreta/plumas/pausa segun  *)
(* S1_Action, aunque S1_CountPreset sea mayor que cero.         *)
(* En paralelo se incrementa el contador y, al llegar al       *)
(* objetivo, se genera S1_CountReachedPulse para las acciones  *)
(* especiales del contador.                                   *)
(* ============================================================ *)

S1_ActionTrigger     := FALSE ;
S1_CountReachedPulse := FALSE ;

S1_Rising :=
    S1_Det AND NOT S1_DetPrev ;


IF (S1_Enable <> 0) AND
   NOT SystemStopActive THEN


    (* ---------- CADA NUEVA DETECCION ---------- *)

    IF S1_Rising THEN

        (* La accion inmediata SIEMPRE ocurre en cada deteccion. *)
        S1_ActionTrigger := TRUE ;

        (* Conteo independiente de la accion inmediata. *)
        IF S1_CountAccum < 32767 THEN
            S1_CountAccum := S1_CountAccum + 1 ;
        END_IF ;

        (* CountPreset>0 dispara una segunda capa de acciones. *)
        IF (S1_CountPreset > 0) AND
           NOT S1_CountDone AND
           (S1_CountAccum >= S1_CountPreset) THEN

            S1_CountDone         := TRUE ;
            S1_CountReachedPulse := TRUE ;

        END_IF ;

    END_IF ;


    (* ---------- INICIAR / RETRIGGER EVENTO INMEDIATO ---------- *)

    IF S1_ActionTrigger THEN

        IF NOT S1_EventActive THEN

            S1_EventActive      := TRUE ;
            S1_EventActive_Reg  := 1 ;
            S1_TimerAccum       := 0 ;

            (* Action 2/4 = evento temporizado.
               Action 0/1/3 = evento sigue presencia fisica. *)
            IF (S1_Action = 2) OR
               (S1_Action = 4) THEN

                S1_WaitActive := TRUE ;

            ELSE

                S1_WaitActive := FALSE ;

            END_IF ;

            (* BandMode=1 permite usar sensor/torreta/plumas sin
               interferir con el movimiento de la banda. *)
            IF (S1_BandMode = 0) AND
               (S1_Action > 0) THEN

                S1_StopLatch := TRUE ;

            ELSE

                S1_StopLatch := FALSE ;

            END_IF ;


        ELSE

            (* Si llega otra pieza mientras un evento temporizado
               sigue activo, reiniciar su ventana temporal. *)
            IF S1_WaitActive THEN
                S1_TimerAccum := 0 ;
            END_IF ;

        END_IF ;

    END_IF ;


    (* ---------- EVENTO QUE SIGUE PRESENCIA DEL SENSOR ---------- *)

    IF S1_EventActive AND
       NOT S1_WaitActive AND
       NOT S1_Det THEN

        S1_EventActive      := FALSE ;
        S1_EventActive_Reg  := 0 ;
        S1_StopLatch        := FALSE ;

    END_IF ;


ELSE

    S1_EventActive      := FALSE ;
    S1_EventActive_Reg  := 0 ;
    S1_StopLatch        := FALSE ;
    S1_WaitActive       := FALSE ;
    S1_TimerAccum       := 0 ;

END_IF ;


S1_DetPrev := S1_Det ;


(* ============================================================ *)
(* §12 - TIMER DEL EVENTO S1                                  *)
(* Action 2/4 usa el timer aunque la banda no este habilitada. *)
(* ============================================================ *)

IF S1_WaitActive AND
   S1_EventActive AND
   NOT SystemStopActive THEN


    IF Pulse1s THEN

        IF S1_TimerAccum < 32767 THEN
            S1_TimerAccum := S1_TimerAccum + 1 ;
        END_IF ;

    END_IF ;


    IF (S1_TimerPreset <= 0) OR
       (S1_TimerAccum >= S1_TimerPreset) THEN

        S1_WaitActive      := FALSE ;
        S1_EventActive     := FALSE ;
        S1_EventActive_Reg := 0 ;
        S1_StopLatch       := FALSE ;
        S1_TimerAccum      := 0 ;

    END_IF ;


ELSIF SystemStopActive THEN

    S1_TimerAccum      := 0 ;
    S1_WaitActive      := FALSE ;
    S1_EventActive     := FALSE ;
    S1_EventActive_Reg := 0 ;
    S1_StopLatch       := FALSE ;

END_IF ;


(* ============================================================ *)
(* §13 - SENSOR S2 COMO EVENTO GENERAL                         *)
(* v5: accion inmediata y contador independientes, igual que S1. *)
(* ============================================================ *)

S2_ActionTrigger     := FALSE ;
S2_CountReachedPulse := FALSE ;

S2_Rising :=
    S2_Det AND NOT S2_DetPrev ;


IF (S2_Enable <> 0) AND
   NOT SystemStopActive THEN


    IF S2_Rising THEN

        S2_ActionTrigger := TRUE ;

        IF S2_CountAccum < 32767 THEN
            S2_CountAccum := S2_CountAccum + 1 ;
        END_IF ;

        IF (S2_CountPreset > 0) AND
           NOT S2_CountDone AND
           (S2_CountAccum >= S2_CountPreset) THEN

            S2_CountDone         := TRUE ;
            S2_CountReachedPulse := TRUE ;

        END_IF ;

    END_IF ;


    IF S2_ActionTrigger THEN

        IF NOT S2_EventActive THEN

            S2_EventActive      := TRUE ;
            S2_EventActive_Reg  := 1 ;
            S2_TimerAccum       := 0 ;

            IF (S2_Action = 2) OR
               (S2_Action = 4) THEN

                S2_WaitActive := TRUE ;

            ELSE

                S2_WaitActive := FALSE ;

            END_IF ;

            IF (S2_BandMode = 0) AND
               (S2_Action > 0) THEN

                S2_StopLatch := TRUE ;

            ELSE

                S2_StopLatch := FALSE ;

            END_IF ;


        ELSE

            IF S2_WaitActive THEN
                S2_TimerAccum := 0 ;
            END_IF ;

        END_IF ;

    END_IF ;


    IF S2_EventActive AND
       NOT S2_WaitActive AND
       NOT S2_Det THEN

        S2_EventActive      := FALSE ;
        S2_EventActive_Reg  := 0 ;
        S2_StopLatch        := FALSE ;

    END_IF ;


ELSE

    S2_EventActive      := FALSE ;
    S2_EventActive_Reg  := 0 ;
    S2_StopLatch        := FALSE ;
    S2_WaitActive       := FALSE ;
    S2_TimerAccum       := 0 ;

END_IF ;


S2_DetPrev := S2_Det ;


(* ============================================================ *)
(* §14 - TIMER DEL EVENTO S2                                  *)
(* ============================================================ *)

IF S2_WaitActive AND
   S2_EventActive AND
   NOT SystemStopActive THEN


    IF Pulse1s THEN

        IF S2_TimerAccum < 32767 THEN
            S2_TimerAccum := S2_TimerAccum + 1 ;
        END_IF ;

    END_IF ;


    IF (S2_TimerPreset <= 0) OR
       (S2_TimerAccum >= S2_TimerPreset) THEN

        S2_WaitActive      := FALSE ;
        S2_EventActive     := FALSE ;
        S2_EventActive_Reg := 0 ;
        S2_StopLatch       := FALSE ;
        S2_TimerAccum      := 0 ;

    END_IF ;


ELSIF SystemStopActive THEN

    S2_TimerAccum      := 0 ;
    S2_WaitActive      := FALSE ;
    S2_EventActive     := FALSE ;
    S2_EventActive_Reg := 0 ;
    S2_StopLatch       := FALSE ;

END_IF ;


(* ============================================================ *)
(* §14b - PARO AUTOMATICO DE BANDA                            *)
(*                                                            *)
(* AutoStopMode 0 = off                                        *)
(* AutoStopMode 1 = cuenta solo tiempo de movimiento efectivo  *)
(* AutoStopMode 2 = cuenta desde START aunque sensor la pause   *)
(* ============================================================ *)

AutoStopRunIntent :=
    MotionCfgPresent AND
    BandEnable AND
    CfgReady AND
    CfgValid AND
    (InitSeqState = 0) AND
    NOT GenStop AND
    NOT CountBandStopLatch AND
    NOT S1_StopLatch AND
    NOT S2_StopLatch AND
    ((DirCmd = 1) OR (DirCmd = 2)) ;


IF BandEnable AND
   (AutoStopMode > 0) AND
   (AutoStopPreset > 0) AND
   NOT GenStop THEN

    IF Pulse1s THEN

        IF (AutoStopMode = 2) OR
           ((AutoStopMode = 1) AND AutoStopRunIntent) THEN

            IF AutoStopAccum < 32767 THEN
                AutoStopAccum := AutoStopAccum + 1 ;
            END_IF ;

        END_IF ;

    END_IF ;


    IF AutoStopAccum >= AutoStopPreset THEN

        BandEnable       := FALSE ;
        BandRunning      := FALSE ;

        AutoStopDone     := TRUE ;
        AutoStopDone_Reg := 1 ;

        StopReason       := 4 ;

    END_IF ;


ELSIF AutoStopMode = 0 THEN

    AutoStopAccum    := 0 ;
    AutoStopDone     := FALSE ;
    AutoStopDone_Reg := 0 ;

END_IF ;


IF AutoStopDone THEN
    AutoStopDone_Reg := 1 ;
ELSE
    AutoStopDone_Reg := 0 ;
END_IF ;


(* Si no hay paro fisico/software, reportar pausas por sensor. *)

IF NOT GenStop THEN

    IF CountBandStopLatch THEN

        IF CountStopSource = 2 THEN
            StopReason := 8 ;
        ELSE
            StopReason := 7 ;
        END_IF ;

    ELSIF AutoStopDone THEN

        StopReason := 4 ;

    ELSIF S2_StopLatch THEN

        StopReason := 6 ;

    ELSIF S1_StopLatch THEN

        StopReason := 5 ;

    ELSE

        StopReason := 0 ;

    END_IF ;

END_IF ;


(* ============================================================ *)
(* §14c - ACCIONES AL ALCANZAR EL CONTEO - COMBINADOR v5      *)
(*                                                            *)
(* La accion inmediata del sensor YA ocurrio independientemente. *)
(* Este bloque es una SEGUNDA CAPA que solo se dispara cuando  *)
(* CountPreset>0 alcanza el objetivo.                          *)
(*                                                            *)
(* ActionMask: 1=pausa banda, 2=stop proceso, 4=luces,         *)
(* 8=direccion, 16=pluma1, 32=pluma2. Se pueden sumar.         *)
(*                                                            *)
(* HoldMode=0: pausa/luz/plumas quedan enclavadas hasta R5/R6. *)
(* HoldMode=1: pausa/luz/plumas duran HoldPreset segundos y    *)
(* despues se liberan automaticamente.                         *)
(*                                                            *)
(* Stop proceso (bit 2) SIEMPRE queda enclavado hasta R5/R6.  *)
(* Cambio de direccion (bit 8) es persistente y se hace con    *)
(* STOP previo cuando la banda estaba corriendo.               *)
(* ============================================================ *)

CountRequestedDir := 0 ;


(* ---------- DISPARO DEL CONTADOR S1 ---------- *)

IF S1_CountReachedPulse AND
   (S1_CountActionMask <> 0) THEN

    CountActionSource_Reg := 1 ;
    S1_CountHoldAccum     := 0 ;

    (* Determinar si existe alguna accion temporal/enclavable. *)
    S1_CountActionActive :=
        S1_CountDoBandStop OR
        S1_CountDoLamp OR
        S1_CountDoPluma1 OR
        S1_CountDoPluma2 ;

    IF S1_CountDoBandStop THEN
        S1_CountBandPauseActive := TRUE ;
        CountStopSource         := 1 ;
    END_IF ;

    IF S1_CountDoSystemStop THEN
        CountSystemStopLatch := TRUE ;
        CountStopSource      := 1 ;
    END_IF ;

    IF S1_CountDoLamp THEN
        S1_CountLampActiveMask := S1_CountLampMask ;
    END_IF ;

    IF S1_CountDoPluma1 THEN
        S1_CountP1Active     := TRUE ;
        S1_CountP1LatchedCmd := S1_CountPluma1Cmd ;
    END_IF ;

    IF S1_CountDoPluma2 THEN
        S1_CountP2Active     := TRUE ;
        S1_CountP2LatchedCmd := S1_CountPluma2Cmd ;
    END_IF ;

    IF S1_CountDoDir THEN

        IF S1_CountDirTarget = 3 THEN

            IF DirCmd = 1 THEN
                CountRequestedDir := 2 ;
            ELSIF DirCmd = 2 THEN
                CountRequestedDir := 1 ;
            END_IF ;

        ELSE

            CountRequestedDir := S1_CountDirTarget ;

        END_IF ;

    END_IF ;

END_IF ;


(* ---------- DISPARO DEL CONTADOR S2 ---------- *)
(* S2 se procesa despues y gana prioridad si ambos llegan en   *)
(* el mismo scan para direccion/plumas.                        *)

IF S2_CountReachedPulse AND
   (S2_CountActionMask <> 0) THEN

    CountActionSource_Reg := 2 ;
    S2_CountHoldAccum     := 0 ;

    S2_CountActionActive :=
        S2_CountDoBandStop OR
        S2_CountDoLamp OR
        S2_CountDoPluma1 OR
        S2_CountDoPluma2 ;

    IF S2_CountDoBandStop THEN
        S2_CountBandPauseActive := TRUE ;
        CountStopSource         := 2 ;
    END_IF ;

    IF S2_CountDoSystemStop THEN
        CountSystemStopLatch := TRUE ;
        CountStopSource      := 2 ;

Sig[2]  := DebFilt[1] ;
Sig[3]  := DebFilt[2] ;
Sig[4]  := DebFilt[3] ;
Sig[5]  := DebFilt[4] ;
Sig[6]  := (MotorState = 1) ;
Sig[7]  := (MotorState <> 1) ;
Sig[8]  := (DirActual = 1) ;
Sig[9]  := (DirActual = 2) ;
Sig[10] := (MotorState = 2) ;
Sig[11] := CfgReady ;
Sig[12] := (espd < 50) ;
Sig[13] := (EC[11] <> 0) ;
Sig[14] := (EC[12] <> 0) ;
Sig[15] := TRUE ;


(* ============================================================ *)
(* §E3 - CONDICIONES C1..C8                                     *)
(* EC[30+6k]: Op (0 off, 1 AND, 2 OR), Src1..Src4, Edge        *)
(* (0 nivel, 1 flanco de subida, 2 flanco de bajada).           *)
(* Una condicion puede usar otra (C1 dentro de C2).            *)
(* ============================================================ *)

FOR ek := 0 TO 7 DO

    eb  := 30 + ek * 6 ;
    eop := EC[eb] ;

    IF eop = 0 THEN
        eres := FALSE ;
    ELSE
        eres  := (eop = 1) ;
        eused := FALSE ;
        FOR ej := 1 TO 4 DO
            esc := EC[eb + ej] ;
            IF esc <> 0 THEN
                eused := TRUE ;
                esn := FALSE ;
                IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
                IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
                IF esn THEN esv := NOT esv ; END_IF ;
                IF eop = 1 THEN
                    eres := eres AND esv ;
                ELSE
                    eres := eres OR esv ;
                END_IF ;
            END_IF ;
        END_FOR ;
        IF NOT eused THEN
            eres := FALSE ;
        END_IF ;
    END_IF ;

    eedge := EC[eb + 5] ;
    CASE eedge OF
        1: eout := eres AND NOT CondPrev[ek] AND NOT EngFirst ;
        2: eout := (NOT eres) AND CondPrev[ek] AND NOT EngFirst ;
        ELSE eout := eres ;
    END_CASE ;

    CondPrev[ek] := eres ;
    Sig[16 + ek] := eout ;

END_FOR ;


(* ============================================================ *)
(* §E4 - TIMERS T1..T8  (base 100 ms, preset en decimas)        *)
(* EC[78+4k]: Type, In, Reset, Preset                           *)
(*   1 TON     : DN si In sigue activo Preset; al caer In -> 0  *)
(*   2 TOF     : DN mientras In y Preset despues de que cae     *)
(*   3 PULSO   : flanco de In -> DN durante Preset              *)
(*   4 RETARDO : flanco de In -> DN al cumplir Preset; DN se    *)
(*               queda hasta Reset o hasta el siguiente flanco  *)
(*               (la caja puede irse antes: el retardo sigue).  *)
(* ============================================================ *)

FOR ek := 0 TO 7 DO

    eb   := 78 + ek * 4 ;
    etyp := EC[eb] ;
    epre := EC[eb + 3] ;

    esc := EC[eb + 1] ;
    esn := FALSE ;
    IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
    IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
    IF esn THEN esv := NOT esv ; END_IF ;
    evin := esv ;

    esc := EC[eb + 2] ;
    esn := FALSE ;
    IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
    IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
    IF esn THEN esv := NOT esv ; END_IF ;
    evrst := esv ;

    erise := evin AND NOT TPrevIn[ek] AND NOT EngFirst ;
    TPrevIn[ek] := evin ;

    IF evrst OR (etyp = 0) THEN

        TAcc[ek] := 0 ;
        TDn[ek]  := FALSE ;
        TRun[ek] := FALSE ;

    ELSIF etyp = 1 THEN

        IF evin THEN
            IF Pulse100ms AND (TAcc[ek] < epre) THEN
                TAcc[ek] := TAcc[ek] + 1 ;
            END_IF ;
            TDn[ek] := (TAcc[ek] >= epre) ;
        ELSE
            TAcc[ek] := 0 ;
            TDn[ek]  := FALSE ;
        END_IF ;

    ELSIF etyp = 2 THEN

        IF evin THEN
            TAcc[ek] := 0 ;
            TDn[ek]  := TRUE ;
            TRun[ek] := TRUE ;
        ELSIF TRun[ek] THEN
            IF Pulse100ms AND (TAcc[ek] < epre) THEN
                TAcc[ek] := TAcc[ek] + 1 ;
            END_IF ;
            IF TAcc[ek] >= epre THEN
                TRun[ek] := FALSE ;
                TDn[ek]  := FALSE ;
            END_IF ;
        ELSE
            TDn[ek] := FALSE ;
        END_IF ;

    ELSIF etyp = 3 THEN

        IF erise AND NOT TRun[ek] THEN
            TRun[ek] := TRUE ;
            TAcc[ek] := 0 ;
        END_IF ;
        IF TRun[ek] THEN
            IF Pulse100ms AND (TAcc[ek] < epre) THEN
                TAcc[ek] := TAcc[ek] + 1 ;
            END_IF ;
            IF TAcc[ek] >= epre THEN
                TRun[ek] := FALSE ;
                TDn[ek]  := FALSE ;
            ELSE
                TDn[ek]  := TRUE ;
            END_IF ;
        ELSE
            TDn[ek] := FALSE ;
        END_IF ;

    ELSIF etyp = 4 THEN

        IF erise THEN
            TRun[ek] := TRUE ;
            TAcc[ek] := 0 ;
            TDn[ek]  := FALSE ;
        END_IF ;
        IF TRun[ek] THEN
            IF Pulse100ms AND (TAcc[ek] < epre) THEN
                TAcc[ek] := TAcc[ek] + 1 ;
            END_IF ;
            IF TAcc[ek] >= epre THEN
                TRun[ek] := FALSE ;
                TDn[ek]  := TRUE ;
            END_IF ;
        END_IF ;

    END_IF ;

    Sig[24 + ek] := TDn[ek] ;

END_FOR ;


(* ============================================================ *)
(* §E5 - CONTADORES CT1..CT8                                    *)
(* EC[110+4k]: In, Reset, Preset, Mode                          *)
(*   Mode 0 = una vez: DN se queda hasta Reset                  *)
(*   Mode 1 = ciclico: pulso DN de un scan cada Preset y        *)
(*            vuelve a 0 (cada N piezas)                        *)
(* Cuenta flancos de subida de CUALQUIER senal.                 *)
(* ============================================================ *)

FOR ek := 0 TO 7 DO

    eb    := 110 + ek * 4 ;
    epre  := EC[eb + 2] ;
    emode := EC[eb + 3] ;

    esc := EC[eb] ;
    esn := FALSE ;
    IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
    IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
    IF esn THEN esv := NOT esv ; END_IF ;
    evin := esv ;

    esc := EC[eb + 1] ;
    esn := FALSE ;
    IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
    IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
    IF esn THEN esv := NOT esv ; END_IF ;
    evrst := esv ;

    erise := evin AND NOT CPrevIn[ek] AND NOT EngFirst ;
    CPrevIn[ek] := evin ;

    IF evrst OR (EC[eb] = 0) THEN

        CAcc[ek] := 0 ;
        CDn[ek]  := FALSE ;

    ELSE

        IF emode = 1 THEN
            CDn[ek] := FALSE ;
        END_IF ;

        IF erise THEN
            IF CAcc[ek] < 32767 THEN
                CAcc[ek] := CAcc[ek] + 1 ;
            END_IF ;
            IF (epre > 0) AND (CAcc[ek] >= epre) THEN
                CDn[ek] := TRUE ;
                IF emode = 1 THEN
                    CAcc[ek] := 0 ;
                END_IF ;
            END_IF ;
        END_IF ;

    END_IF ;

    Sig[32 + ek] := CDn[ek] ;

END_FOR ;


(* ============================================================ *)
(* §E6 - MEMORIAS M1..M8 (set / reset)                          *)
(* EC[142+3k]: Set, Reset, Prio (0 gana reset, 1 gana set)      *)
(* "Mantener una luz hasta que ocurra otra condicion".          *)
(* ============================================================ *)

FOR ek := 0 TO 7 DO

    eb := 142 + ek * 3 ;

    IF (EC[eb] = 0) AND (EC[eb + 1] = 0) THEN

        MBit[ek] := FALSE ;

    ELSE

        esc := EC[eb] ;
        esn := FALSE ;
        IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
        IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
        IF esn THEN esv := NOT esv ; END_IF ;
        evin := esv ;

        esc := EC[eb + 1] ;
        esn := FALSE ;
        IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
        IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
        IF esn THEN esv := NOT esv ; END_IF ;
        evrst := esv ;

        IF EC[eb + 2] = 1 THEN
            IF evin THEN
                MBit[ek] := TRUE ;
            ELSIF evrst THEN
                MBit[ek] := FALSE ;
            END_IF ;
        ELSE
            IF evrst THEN
                MBit[ek] := FALSE ;
            ELSIF evin THEN
                MBit[ek] := TRUE ;
            END_IF ;
        END_IF ;

    END_IF ;

    Sig[40 + ek] := MBit[ek] ;

END_FOR ;


(* ============================================================ *)
(* §E7 - SECUENCIA (MAQUINA DE ESTADOS, hasta 16 pasos)         *)
(* EC[166..170]: Start, Stop, Reset, Mode (0 una vez, 1 ciclo), *)
(*               StepCount                                      *)
(* EC[172+5k]  : Actions, TransMode, TransSrc, Time, Next       *)
(*   TransMode 0 tiempo · 1 senal (nivel) · 2 tiempo Y senal    *)
(*             3 senal O tiempo limite · 4 flanco de la senal   *)
(*   Next 0 = el paso siguiente; 1..16 = salto a ese paso.      *)
(* Actions (bits): 0 verde 1 amarilla 2 roja 3 P1 sube          *)
(*   4 P1 baja 5 P2 sube 6 P2 baja 7 marcha 8 paro 9 dir 1      *)
(*   10 dir 2 11 invertir (al entrar al paso)                   *)
(* Cualquier paro (§4) aborta la secuencia: el motor nunca     *)
(* rearranca solo al soltar un paro.                            *)
(* ============================================================ *)

ecnt := EC[170] ;

esc := EC[166] ;
esn := FALSE ;
IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
IF esn THEN esv := NOT esv ; END_IF ;
eSeqStartV := esv ;

esc := EC[167] ;
esn := FALSE ;
IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
IF esn THEN esv := NOT esv ; END_IF ;
eSeqStopV := esv ;

esc := EC[168] ;
esn := FALSE ;
IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
IF esn THEN esv := NOT esv ; END_IF ;
eSeqResetV := esv ;

StepEntry := FALSE ;

IF (ecnt < 1) OR (ecnt > 16) OR eSeqResetV OR eSeqStopV OR
   GenStop OR EngFirst THEN

    ESeqActive := FALSE ;
    ESeqStep   := 0 ;
    ESeqAcc    := 0 ;

ELSE

    IF eSeqStartV AND NOT ESeqStartPrev AND NOT ESeqActive THEN

        ESeqActive := TRUE ;
        ESeqStep   := 1 ;
        ESeqAcc    := 0 ;
        StepEntry := TRUE ;

    ELSIF ESeqActive THEN

        IF (ESeqStep < 1) OR (ESeqStep > ecnt) THEN
            ESeqStep := 1 ;
        END_IF ;

        esb := 172 + (ESeqStep - 1) * 5 ;
        etm := EC[esb + 1] ;
        est := EC[esb + 3] ;

        esc := EC[esb + 2] ;
        esn := FALSE ;
        IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
        IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
        IF esn THEN esv := NOT esv ; END_IF ;
        etv := esv ;

        IF Pulse100ms AND (ESeqAcc < 32767) THEN
            ESeqAcc := ESeqAcc + 1 ;
        END_IF ;

        etimeOk := (ESeqAcc >= est) ;
        etRise  := etv AND NOT ESeqTransPrev ;
        ESeqTransPrev := etv ;

        CASE etm OF
            0: edone := etimeOk ;
            1: edone := etv ;
            2: edone := etimeOk AND etv ;
            3: edone := etv OR ((est > 0) AND etimeOk) ;
            4: edone := etRise ;
            ELSE edone := FALSE ;
        END_CASE ;

        IF edone THEN

            enx := EC[esb + 4] ;
            IF enx <= 0 THEN
                enx := ESeqStep + 1 ;
            END_IF ;

            IF enx > ecnt THEN
                IF EC[169] = 1 THEN
                    enx := 1 ;
                ELSE
                    enx := 0 ;
                END_IF ;
            END_IF ;

            IF enx = 0 THEN
                ESeqActive := FALSE ;
                ESeqStep   := 0 ;
            ELSE
                ESeqStep   := enx ;
                StepEntry := TRUE ;
            END_IF ;

            ESeqAcc := 0 ;

        END_IF ;

    END_IF ;

    (* Al entrar a un paso, una senal que YA estaba activa no cuenta
       como flanco: TransMode 4 espera un evento NUEVO. *)
    IF StepEntry THEN
        esb := 172 + (ESeqStep - 1) * 5 ;
        esc := EC[esb + 2] ;
        esn := FALSE ;
        IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
        IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
        IF esn THEN esv := NOT esv ; END_IF ;
        ESeqTransPrev := esv ;
    END_IF ;

END_IF ;

ESeqStartPrev := eSeqStartV ;

FOR ek := 0 TO 15 DO
    Sig[48 + ek] := ESeqActive AND (ESeqStep = ek + 1) ;
END_FOR ;

IF ESeqActive THEN
    eacts := EC[172 + (ESeqStep - 1) * 5] ;
ELSE
    eacts := 0 ;
END_IF ;

FOR ej := 0 TO 11 DO
    ESeqBit[ej] := ((eacts MOD 2) = 1) ;
    eacts := eacts / 2 ;
END_FOR ;


(* ============================================================ *)
(* §E8 - MOTOR: UNA SOLA MAQUINA DE ESTADOS                     *)
(*                                                              *)
(*  0 PARADO ---(quiere marcha)---------------> 1 MARCHA        *)
(*  1 MARCHA ---(cambia la direccion pedida)--> 2 FRENANDO      *)
(*  2 FRENANDO -(t >= EC[8] Y VFD en cero)----> 1 en la nueva   *)
(*  cualquier paro / config no lista ---------> 0               *)
(*                                                              *)
(* Fuentes: EC[1] arranque, EC[2] paro, EC[3] modo (0 enclavado *)
(* 1 mientras), EC[4] pausa, EC[5] dir 1, EC[6] dir 2,          *)
(* EC[7] invertir. Prioridad: paro > pausa > direccion > marcha *)
(* DirCmd (backend) es SOLO la direccion inicial: la direccion  *)
(* real (DirActual) la escribe UNICAMENTE esta seccion.         *)
(* Nunca invierte con menos de 1 s ni sin confirmar velocidad   *)
(* cero (o 5 s extra si el VFD no reporta).                     *)
(* ============================================================ *)

FOR ek := 0 TO 5 DO
    CASE ek OF
        0: esc := EC[1] ;
        1: esc := EC[2] ;
        2: esc := EC[4] ;
        3: esc := EC[5] ;
        4: esc := EC[6] ;
        ELSE esc := EC[7] ;
    END_CASE ;
    esn := FALSE ;
    IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
    IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
    IF esn THEN esv := NOT esv ; END_IF ;
    HdrV[ek] := esv ;
END_FOR ;

erunRise := HdrV[0] AND NOT RunPrev AND NOT EngFirst ;
efwdRise := HdrV[3] AND NOT FwdPrev AND NOT EngFirst ;
erevRise := HdrV[4] AND NOT RevPrev AND NOT EngFirst ;
etogRise := HdrV[5] AND NOT TogPrev AND NOT EngFirst ;
RunPrev := HdrV[0] ;
FwdPrev := HdrV[3] ;
RevPrev := HdrV[4] ;
TogPrev := HdrV[5] ;

emotorOK := MotionCfgPresent AND CfgReady AND CfgValid AND
            (InitSeqState = 0) AND NOT GenStop ;

IF NOT emotorOK THEN
    RunLatch := FALSE ;
ELSIF EC[3] = 1 THEN
    RunLatch := HdrV[0] AND NOT HdrV[1] ;
ELSIF HdrV[1] THEN
    RunLatch := FALSE ;
ELSIF erunRise THEN
    RunLatch := TRUE ;
END_IF ;

IF efwdRise THEN
    DirTarget := 1 ;
END_IF ;
IF erevRise THEN
    DirTarget := 2 ;
END_IF ;
IF etogRise THEN
    IF DirTarget = 1 THEN DirTarget := 2 ; ELSE DirTarget := 1 ; END_IF ;
END_IF ;

IF ESeqActive THEN
    IF ESeqBit[9] THEN
        DirTarget := 1 ;
    END_IF ;
    IF ESeqBit[10] THEN
        DirTarget := 2 ;
    END_IF ;
    IF ESeqBit[11] AND StepEntry THEN
        IF DirTarget = 1 THEN DirTarget := 2 ; ELSE DirTarget := 1 ; END_IF ;
    END_IF ;
END_IF ;

ESeqExcl  := (EC[15] <> 0) ;
eseqRun  := ESeqActive AND ESeqBit[7] ;
eseqStop := ESeqActive AND ESeqBit[8] ;

IF ESeqActive AND ESeqExcl THEN
    ewantRun := eseqRun AND NOT eseqStop ;
ELSE
    ewantRun := (RunLatch OR eseqRun) AND NOT eseqStop ;
END_IF ;

IF HdrV[2] OR NOT emotorOK THEN
    ewantRun := FALSE ;
END_IF ;

RevT := EC[8] ;
IF RevT < 10 THEN
    RevT := 10 ;
END_IF ;

IF (NOT emotorOK) AND (MotorState <> 0) THEN
    MotorState := 0 ;
    BrakeAcc   := 0 ;
END_IF ;

IF Pulse100ms AND (MotorState <> 1) AND (BrakeAcc < 32767) THEN
    BrakeAcc := BrakeAcc + 1 ;
END_IF ;

ebrakeOk := (BrakeAcc >= RevT) AND ((espd < 50) OR (BrakeAcc >= RevT + 50)) ;

IF MotorState = 0 THEN

    IF (DirTarget <> DirActual) AND ebrakeOk THEN
        DirActual := DirTarget ;
    END_IF ;
    IF ewantRun AND (DirTarget = DirActual) THEN
        MotorState := 1 ;
    END_IF ;

ELSIF MotorState = 1 THEN

    IF NOT ewantRun THEN
        MotorState := 0 ;
        BrakeAcc   := 0 ;
    ELSIF DirTarget <> DirActual THEN
        MotorState := 2 ;
        BrakeAcc   := 0 ;
    END_IF ;

ELSIF MotorState = 2 THEN

    IF ebrakeOk THEN
        DirActual := DirTarget ;
        IF ewantRun THEN
            MotorState := 1 ;
        ELSE
            MotorState := 0 ;
        END_IF ;
    ELSIF NOT ewantRun THEN
        MotorState := 0 ;
    END_IF ;

ELSE

    MotorState := 0 ;

END_IF ;

IF MotorState = 1 THEN
    IF DirActual = 2 THEN
        VFD_Control := 34 ;
    ELSE
        VFD_Control := 18 ;
    END_IF ;
    BandRunning := TRUE ;
ELSE
    VFD_Control := 1 ;
    BandRunning := FALSE ;
END_IF ;

BandEnable := RunLatch ;

IF NOT GenStop THEN
    StopReason := 0 ;
END_IF ;


(* ============================================================ *)
(* §E9 - SALIDAS: TORRETA Y PLUMAS                              *)
(* EC[20..26] fuente de verde, amarilla, roja, P1 sube, P1 baja,*)
(* P2 sube, P2 baja. Con la secuencia activa se suman sus bits  *)
(* (o mandan solos si EC[15] = exclusiva).                      *)
(* Pluma: subir Y bajar a la vez -> ambas apagadas (falla).     *)
(* EC[13] = tiempo maximo de movimiento de una pluma (0.1 s).   *)
(* R60/R61 (manual del panel) solo si el motor no pide nada.    *)
(* I3: plumas apagadas y solo las luces de EC[14].              *)
(* ============================================================ *)

FOR ek := 0 TO 6 DO
    esc := EC[20 + ek] ;
    esn := FALSE ;
    IF esc >= 100 THEN esn := TRUE ; esc := esc - 100 ; END_IF ;
    IF (esc >= 0) AND (esc <= 63) THEN esv := Sig[esc] ; ELSE esv := FALSE ; END_IF ;
    IF esn THEN esv := NOT esv ; END_IF ;
    IF ESeqActive AND ESeqExcl THEN
        OutReq[ek] := ESeqBit[ek] ;
    ELSIF ESeqActive THEN
        OutReq[ek] := esv OR ESeqBit[ek] ;
    ELSE
        OutReq[ek] := esv ;
    END_IF ;
END_FOR ;

PlumaFault := 0 ;

FOR ek := 0 TO 1 DO

    eup := OutReq[3 + ek * 2] ;
    edn := OutReq[4 + ek * 2] ;

    IF eup AND edn THEN
        ecmd := 0 ;
        IF ek = 0 THEN PlumaFault := PlumaFault + 1 ; ELSE PlumaFault := PlumaFault + 4 ; END_IF ;
    ELSIF eup THEN
        ecmd := 1 ;
    ELSIF edn THEN
        ecmd := 2 ;
    ELSE
        ecmd := 0 ;
    END_IF ;

    IF (ecmd = 0) AND NOT (eup AND edn) THEN
        IF ek = 0 THEN
            IF (Pluma1Cmd = 1) OR (Pluma1Cmd = 2) THEN ecmd := Pluma1Cmd ; END_IF ;
        ELSE
            IF (Pluma2Cmd = 1) OR (Pluma2Cmd = 2) THEN ecmd := Pluma2Cmd ; END_IF ;
        END_IF ;
    END_IF ;

    IF ecmd <> PlPrev[ek] THEN
        PlAcc[ek]     := 0 ;
        PlTimeout[ek] := FALSE ;
    END_IF ;
    PlPrev[ek] := ecmd ;

    IF ecmd <> 0 THEN
        IF Pulse100ms AND (PlAcc[ek] < 32767) THEN
            PlAcc[ek] := PlAcc[ek] + 1 ;
        END_IF ;
        IF (EC[13] > 0) AND (PlAcc[ek] >= EC[13]) THEN
            PlTimeout[ek] := TRUE ;
        END_IF ;
    END_IF ;

    IF PlTimeout[ek] THEN
        ecmd := 0 ;
        IF ek = 0 THEN PlumaFault := PlumaFault + 2 ; ELSE PlumaFault := PlumaFault + 8 ; END_IF ;
    END_IF ;

    PlCmd[ek] := ecmd ;

END_FOR ;

elampG := OutReq[0] ;
elampY := OutReq[1] ;
elampR := OutReq[2] ;

IF SystemStopActive THEN
    ehm    := EC[14] ;
    elampG := ((ehm MOD 2) = 1) ;
    elampY := (((ehm / 2) MOD 2) = 1) ;
    elampR := (((ehm / 4) MOD 2) = 1) ;
    PlCmd[0] := 0 ;
    PlCmd[1] := 0 ;
END_IF ;

PhQ3 := elampG ;
PhQ4 := elampY ;
PhQ5 := elampR ;

PhQ8 := (PlCmd[0] = 1) ;
PhQ9 := (PlCmd[0] = 2) ;
PhQ6 := (PlCmd[1] = 1) ;
PhQ7 := (PlCmd[1] = 2) ;

Pluma1Status := PlCmd[0] ;
Pluma2Status := PlCmd[1] ;

tVerde    := elampG ;
tAmarilla := elampY ;
tRoja     := elampR ;


(* ============================================================ *)
(* §E10 - ESTADO DEL MOTOR -> ES (%R460..%R499)                 *)
(* ============================================================ *)

ES[3]  := MotorState ;
ES[4]  := DirActual ;
ES[5]  := DirTarget ;
IF RunLatch THEN ES[6] := 1 ; ELSE ES[6] := 0 ; END_IF ;
IF ESeqActive THEN ES[7] := 1 ; ELSE ES[7] := 0 ; END_IF ;
ES[8]  := ESeqStep ;
ES[9]  := ESeqAcc ;

emC := 0 ; emT := 0 ; emK := 0 ; emM := 0 ;
ebit := 1 ;
FOR ek := 0 TO 7 DO
    IF Sig[16 + ek] THEN emC := emC + ebit ; END_IF ;
    IF TDn[ek]      THEN emT := emT + ebit ; END_IF ;
    IF CDn[ek]      THEN emK := emK + ebit ; END_IF ;
    IF MBit[ek]     THEN emM := emM + ebit ; END_IF ;
    ES[16 + ek] := TAcc[ek] ;
    ES[24 + ek] := CAcc[ek] ;
    ebit := ebit * 2 ;
END_FOR ;
ES[10] := emC ;
ES[11] := emT ;
ES[12] := emK ;
ES[13] := emM ;

emO := 0 ;
IF PhQ3 THEN emO := emO + 1 ; END_IF ;
IF PhQ4 THEN emO := emO + 2 ; END_IF ;
IF PhQ5 THEN emO := emO + 4 ; END_IF ;
IF PhQ8 THEN emO := emO + 8 ; END_IF ;
IF PhQ9 THEN emO := emO + 16 ; END_IF ;
IF PhQ6 THEN emO := emO + 32 ; END_IF ;
IF PhQ7 THEN emO := emO + 64 ; END_IF ;
ES[14] := emO ;

emI := 0 ;
ebit := 1 ;
FOR ek := 0 TO 4 DO
    IF DebFilt[ek] THEN emI := emI + ebit ; END_IF ;
    ebit := ebit * 2 ;
END_FOR ;
ES[15] := emI ;
